def git(*a): return subprocess.run(["git", *a], check=True, text=True, capture_output=True).stdout

def reliability_problems(files):
    out = []
    for f in files:
        for c in tomllib.loads(Path(f).read_text()).get("claim", []):
            i = c.get("interpretation") or {}
            reviewed = bool(i) or bool(c.get("placeholders"))
            if "reliability" in i:
                if not str(i.get("reason", "")).strip():
                    out.append((str(f), c["id"], "reliability without reason"))
                kinds = {e.get("kind") for e in c.get("evidence", [])}
                if not kinds & OK_EVIDENCE:
                    out.append((str(f), c["id"], f"reliability without probe/document evidence {kinds}"))
            if reviewed and c.get("reviewed_by") != "claude":
                out.append((str(f), c["id"], f"reviewed_by={c.get('reviewed_by')!r}"))
    return out

print("\n## Shared-file isolation (ruling 82)")
def toml_at(rev, p): return tomllib.loads(git("show", f"{rev}:{p}"))
def json_at(rev, p): return json.loads(git("show", f"{rev}:{p}"))
RO = "src/opendata_atlas/portals/registry_overrides.toml"
VJ = "src/opendata_atlas/portals/validation.json"
def overrides_violations(base, head):
    keys = set(base) | set(head)
    return sorted(k for k in keys if k != PACK and base.get(k) != head.get(k))
def validation_violations(base, head):
    out = [k for k in set(base) | set(head) if k != "portals" and base.get(k) != head.get(k)]
    bp, hp = base.get("portals", {}), head.get("portals", {})
    out += [f"portals[{k}]" for k in set(bp) | set(hp) if k != PACK and bp.get(k) != hp.get(k)]
    return sorted(out)
ob, oh = toml_at(BASE, RO), toml_at("HEAD", RO)
vb, vh = json_at(BASE, VJ), json_at("HEAD", VJ)
# known positives: one Seattle validation entry's outcome changed; one other override table changed
kp = copy.deepcopy(vb)
seattle = kp["portals"]["data.seattle.gov"]
def flip(o):
    if isinstance(o, dict):
        if "outcome" in o: o["outcome"] = "fail" if o["outcome"] != "fail" else "pass"; return True
        return any(flip(v) for v in o.values())
    if isinstance(o, list): return any(flip(v) for v in o)
    return False
assert flip(seattle)
print("known positive validation.json:", validation_violations(vb, kp))
