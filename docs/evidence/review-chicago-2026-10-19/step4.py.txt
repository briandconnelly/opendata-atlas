import asyncio, json
from pathlib import Path
from fastmcp import Client
from opendata_atlas.mcp.server import build_server
from opendata_atlas.portals import freshness
from opendata_atlas.portals.knowledge import pack_levels
from opendata_atlas.portals.loader import load_portal
PD = Path("src/opendata_atlas/portals/data.cityofchicago.org")
async def call(c, name, args):
    r = await c.call_tool_mcp(name, args)
    return r.model_dump(mode="json", by_alias=True, exclude_none=True)
async def main():
    out = {}
    t = freshness.today()
    levels = pack_levels(load_portal(PD), today=t)
    out["today"] = str(t); out["pack_levels"] = levels
    async with Client(build_server()) as c:
        r = await call(c, "atlas_find_portals", {"query": "Chicago"})
        s = r["structuredContent"]
        cands = s.get("candidates") or s.get("portals") or []
        out["find_first_id"] = cands[0]["id"] if cands else None
        out["find_isError"] = r.get("isError")
        out["describe"] = {}
        for ds in levels:
            r = await call(c, "atlas_describe_dataset", {"portal": "chicago", "dataset_id": ds})
            k = (r.get("structuredContent") or {}).get("knowledge") or {}
            out["describe"][ds] = {"isError": r.get("isError"), "level": k.get("level"), "expected": levels[ds], "match": k.get("level") == levels[ds]}
        r = await call(c, "atlas_get_reference", {"topic": "portals", "item_id": "chicago"})
        body = (r.get("structuredContent") or {}).get("body", "")
        out["reference_isError"] = r.get("isError")
        out["reference_tier_line"] = [l for l in body.splitlines() if "tier" in l]
    print(json.dumps(out, indent=1))
asyncio.run(main())
