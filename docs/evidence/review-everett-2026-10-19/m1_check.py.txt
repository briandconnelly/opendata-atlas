"""M1 check on the fixed engine: describe f6vp-3svh neighborhood. Known positive: the same call
with the pre-fix behavior restored in memory (placeholder_share-only declarations) gives 1.0."""
import asyncio, json
from fastmcp import Client
from opendata_atlas.mcp.server import build_server
from opendata_atlas.portals import model

cls = next(v for v in vars(model).values() if isinstance(v, type) and "placeholder_declarations" in vars(v))
fixed = cls.placeholder_declarations
def prefix(self, dataset_id, column):
    return [c for c in self.claims if c.kind == "placeholder_share" and c.scope.dataset == dataset_id
            and c.scope.column == column and c.reviewed]

async def describe(c):
    d = (await c.call_tool_mcp("atlas_describe_dataset", {"portal": "everett", "dataset_id": "f6vp-3svh", "detail": "full"})).model_dump(mode="json", by_alias=True, exclude_none=True)["structuredContent"]
    nb = next(p for p in d["place_fields"] if p["column"] == "neighborhood")
    vals = [v.get("value") if isinstance(v, dict) else v for v in (nb.get("values") or [])]
    return {"assigned_fraction": nb.get("assigned_fraction"), "value_count": nb.get("value_count"),
            "placeholder_values": nb.get("placeholder_values"), "Not Assigned in values": "Not Assigned" in vals,
            "n_values": len(vals), "reliability": nb.get("reliability"), "level": d["knowledge"]["level"]}

def check(r):
    ph = [p["value"] for p in r["placeholder_values"] or []]
    return r["assigned_fraction"] < 1.0 and "Not Assigned" in ph and not r["Not Assigned in values"]

async def main():
    async with Client(build_server()) as c:
        cls.placeholder_declarations = prefix
        kp = await describe(c)
        print("known positive (pre-fix behavior in memory):", json.dumps(kp), "-> resolved:", check(kp))
        cls.placeholder_declarations = fixed
        r = await describe(c)
        print("fixed engine (as on disk):", json.dumps(r), "-> resolved:", check(r))
asyncio.run(main())
