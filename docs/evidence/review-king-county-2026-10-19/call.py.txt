"""In-process tool call helper. Usage: uv run python call.py TOOL JSON [--keys k1,k2]"""
import asyncio, json, sys
from fastmcp import Client
from opendata_atlas.mcp.server import build_server

async def call(name, args):
    async with Client(build_server()) as c:
        r = await c.call_tool_mcp(name, args)
    return r.model_dump(mode="json", by_alias=True, exclude_none=True)

if __name__ == "__main__":
    r = asyncio.run(call(sys.argv[1], json.loads(sys.argv[2])))
    sc = r.get("structuredContent") or {}
    sc.pop("sample_rows", None)  # never print raw rows (personal data)
    print(json.dumps({"isError": r.get("isError"), **sc}, indent=1)[:int(sys.argv[3]) if len(sys.argv) > 3 else 6000])
