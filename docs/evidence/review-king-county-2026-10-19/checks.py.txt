def evidence_problems(pack_dir: Path) -> list[str]:
    out = []
    for path in [*sorted((pack_dir / "facts").glob("*.toml")), pack_dir / "regions.toml"]:
        if not path.is_file():
            continue
        for c in tomllib.loads(path.read_text()).get("claim", []):
            interp = c.get("interpretation") or {}
            kinds = {e.get("kind") for e in c.get("evidence", [])}
            if "reliability" in interp:
                if not interp.get("reason"):
                    out.append(f"{path.name}:{c['id']}: reliability without reason")
                if not kinds & {"soql_probe", "metadata_probe", "document"}:
                    out.append(f"{path.name}:{c['id']}: reliability without probe/document")
            if (interp or c.get("placeholders")) and c.get("reviewed_by") != "claude":
                out.append(f"{path.name}:{c['id']}: reviewed_by={c.get('reviewed_by')!r}")
            if interp and not interp.get("reason"):
                out.append(f"{path.name}:{c['id']}: interpretation without reason")
    return out

def validation_diffs(base: dict, head: dict) -> list[str]:
    out = [f"top-level {k}" for k in set(base) | set(head) if k != "portals" and base.get(k) != head.get(k)]
    bp, hp = base.get("portals", {}), head.get("portals", {})
    out += [f"portals[{k}]" for k in sorted(set(bp) | set(hp)) if bp.get(k) != hp.get(k)]
    return out
